@echo off
title Madium Manager [sky land]
setlocal enabledelayedexpansion

:: Check for admin privileges and elevate if needed
net session >nul 2>&1
if %errorlevel% neq 0 (
    echo ==================================================
    echo    SOLICITANDO PRIVILEGIOS DE ADMINISTRADOR...
    echo ==================================================
    powershell -NoProfile -Command "Start-Process '%~f0' -Verb RunAs"
    exit /b
)

:menu
cls
echo ==================================================
echo              MADIUM MANAGER - MENU
echo ==================================================
echo [1] Baixar e Extrair Madium
echo [2] Deletar arquivos do Madium (Limpeza)
echo [3] Baixar e Instalar Dependencias (Ultra-Check)
echo [4] Baixar e Instalar Fishstrap (+ .NET 6.0)
echo [5] Sair
echo ==================================================
set /p opt="Escolha uma opcao: "

if "%opt%"=="1" goto baixar_madium
if "%opt%"=="2" goto deletar_madium
if "%opt%"=="3" goto dependencias
if "%opt%"=="4" goto baixar_fishstrap
if "%opt%"=="5" exit
goto menu

:baixar_madium
cls
echo ==================================================
echo           BAIXANDO E EXTRAINDO MADIUM
echo ==================================================
set "zip_url=https://files.manuscdn.com/user_upload_by_module/session_file/310519663151269258/hEjTkYUgSVPmZlTE.zip"
set "zip_path=%temp%\Madium.zip"
set "extract_path=%USERPROFILE%\Downloads\Madium"

echo Baixando Madium...
curl -L -o "%zip_path%" "%zip_url%"
if %errorlevel% neq 0 powershell -Command "Invoke-WebRequest -Uri '%zip_url%' -OutFile '%zip_path%'"

echo Extraindo arquivos...
if not exist "%extract_path%" mkdir "%extract_path%"
powershell -Command "Expand-Archive -Path '%zip_path%' -DestinationPath '%extract_path%' -Force"
del "%zip_path%" /f /q >nul 2>&1
echo Pronto! Extraido em: %extract_path%
pause
goto menu

:deletar_madium
cls
echo Limpando arquivos do Madium...
taskkill /F /IM Madium.exe >nul 2>&1
taskkill /F /IM madium.exe >nul 2>&1
taskkill /F /IM Loader.exe >nul 2>&1
timeout /t 1 /nobreak >nul
rmdir /S /Q "%LocalAppData%\Madium" >nul 2>&1
echo Limpeza concluida.
pause
goto menu

:dependencias
cls
echo ==================================================
echo        VERIFICANDO DEPENDENCIAS (100%% CERTEZA)
echo ==================================================

:: --- 1. VERIFICAR .NET RUNTIME 8.0 ---
set "dotnet8_installed=0"
:: Checa via comando dotnet
powershell -Command "dotnet --list-runtimes" | findstr /i "Microsoft.WindowsDesktop.App 8.0" >nul
if %errorlevel% equ 0 set "dotnet8_installed=1"
:: Checa via pasta de sistema (backup)
if %dotnet8_installed% equ 0 (
    if exist "%ProgramFiles%\dotnet\shared\Microsoft.WindowsDesktop.App\8.0*" set "dotnet8_installed=1"
)

:: --- 2. VERIFICAR VISUAL C++ REDIST X64 ---
set "vcredist64_installed=0"
:: Checa no Registro (Vários locais)
reg query "HKLM\SOFTWARE\Microsoft\VisualStudio\14.0\VC\Runtimes\x64" /v "Installed" >nul 2>&1
if %errorlevel% equ 0 set "vcredist64_installed=1"
if %vcredist64_installed% equ 0 (
    reg query "HKLM\SOFTWARE\Classes\Installer\Dependencies\VC,redist.x64,amd64,14.0,bundle" >nul 2>&1
    if %errorlevel% equ 0 set "vcredist64_installed=1"
)
:: Checa via Arquivo de Sistema (DLL Crítica)
if %vcredist64_installed% equ 0 (
    if exist "%SystemRoot%\System32\vcruntime140.dll" set "vcredist64_installed=1"
)

:: --- 3. VERIFICAR VISUAL C++ REDIST X86 ---
set "vcredist86_installed=0"
:: Checa no Registro (Vários locais incluindo WOW6432Node)
reg query "HKLM\SOFTWARE\WOW6432Node\Microsoft\VisualStudio\14.0\VC\Runtimes\x86" /v "Installed" >nul 2>&1
if %errorlevel% equ 0 set "vcredist86_installed=1"
if %vcredist86_installed% equ 0 (
    reg query "HKLM\SOFTWARE\Classes\Installer\Dependencies\VC,redist.x86,x86,14.0,bundle" >nul 2>&1
    if %errorlevel% equ 0 set "vcredist86_installed=1"
)
:: Checa via Arquivo de Sistema (DLL Crítica em SysWOW64)
if %vcredist86_installed% equ 0 (
    if exist "%SystemRoot%\SysWOW64\vcruntime140.dll" set "vcredist86_installed=1"
)

:: --- 4. VERIFICAR WEBVIEW2 ---
set "webview2_installed=0"
:: Checa no Registro (Máquina e Usuário)
reg query "HKLM\SOFTWARE\WOW6432Node\Microsoft\EdgeUpdate\Clients\{F3C4FE00-EFC5-498D-9529-EE5816BB8851}" /v "pv" >nul 2>&1
if %errorlevel% equ 0 set "webview2_installed=1"
if %webview2_installed% equ 0 (
    reg query "HKCU\Software\Microsoft\EdgeUpdate\Clients\{F3C4FE00-EFC5-498D-9529-EE5816BB8851}" /v "pv" >nul 2>&1
    if %errorlevel% equ 0 set "webview2_installed=1"
)
:: Checa via Caminho de Execução
if %webview2_installed% equ 0 (
    if exist "%ProgramFiles(x86)%\Microsoft\EdgeWebView\Application" set "webview2_installed=1"
)

:: --- PROCESSO DE INSTALAÇÃO ---
echo [1/4] .NET Runtime 8.0: 
if %dotnet8_installed% equ 1 (echo    - JA INSTALADO.) else (
    echo    - BAIXANDO...
    curl -L -o "%temp%\dotnet8.exe" "https://builds.dotnet.microsoft.com/dotnet/WindowsDesktop/8.0.27/windowsdesktop-runtime-8.0.27-win-x64.exe"
    start /wait "" "%temp%\dotnet8.exe" /install /quiet /norestart
    del "%temp%\dotnet8.exe"
)

echo [2/4] Visual C++ x64:
if %vcredist64_installed% equ 1 (echo    - JA INSTALADO.) else (
    echo    - BAIXANDO...
    curl -L -o "%temp%\vc64.exe" "https://aka.ms/vc14/vc_redist.x64.exe"
    start /wait "" "%temp%\vc64.exe" /install /quiet /norestart
    del "%temp%\vc64.exe"
)

echo [3/4] Visual C++ x86:
if %vcredist86_installed% equ 1 (echo    - JA INSTALADO.) else (
    echo    - BAIXANDO...
    curl -L -o "%temp%\vc86.exe" "https://aka.ms/vc14/vc_redist.x86.exe"
    start /wait "" "%temp%\vc86.exe" /install /quiet /norestart
    del "%temp%\vc86.exe"
)

echo [4/4] WebView2:
if %webview2_installed% equ 1 (echo    - JA INSTALADO.) else (
    echo    - BAIXANDO...
    curl -L -o "%temp%\wv2.exe" "https://go.microsoft.com/fwlink/?linkid=2124701"
    start /wait "" "%temp%\wv2.exe" /silent /install
    del "%temp%\wv2.exe"
)

echo.
echo Verificacao concluida com 100%% de certeza.
pause
goto menu

:baixar_fishstrap
cls
echo ==================================================
echo            BAIXANDO FISHSTRAP E DEPENDENCIAS
echo ==================================================

:: Verificando .NET Desktop Runtime 6.0 (Ultra-Check)
set "dotnet6_installed=0"
powershell -Command "dotnet --list-runtimes" | findstr /i "Microsoft.WindowsDesktop.App 6.0" >nul
if %errorlevel% equ 0 set "dotnet6_installed=1"
if %dotnet6_installed% equ 0 (
    if exist "%ProgramFiles%\dotnet\shared\Microsoft.WindowsDesktop.App\6.0*" set "dotnet6_installed=1"
)

if %dotnet6_installed% equ 1 (
    echo [OK] .NET Desktop Runtime 6.0 ja instalado.
) else (
    echo [!] Baixando .NET Desktop Runtime 6.0...
    curl -L -o "%temp%\dotnet6.exe" "https://builds.dotnet.microsoft.com/dotnet/WindowsDesktop/6.0.36/windowsdesktop-runtime-6.0.36-win-x64.exe"
    start /wait "" "%temp%\dotnet6.exe" /install /quiet /norestart
    del "%temp%\dotnet6.exe"
)

echo Baixando Fishstrap...
curl -L -o "%temp%\Fishstrap.exe" "https://fishstrap.app/Fishstrap.exe"
start "" "%temp%\Fishstrap.exe"
timeout /t 10 >nul
del "%temp%\Fishstrap.exe" /f /q >nul 2>&1
echo Concluido!
pause
goto menu
